DROP TRIGGER IF EXISTS trigger_log_aluno_exclusao ON ALUNO;
DROP TRIGGER IF EXISTS trigger_log_professor_exclusao ON PROFESSOR;
DROP FUNCTION IF EXISTS salvar_aluno_excluido();
DROP FUNCTION IF EXISTS salvar_professor_excluido();

DROP TABLE IF EXISTS REGISTRO_USO CASCADE;
DROP TABLE IF EXISTS AVALIACAO CASCADE;
DROP TABLE IF EXISTS MATRICULA CASCADE;
DROP TABLE IF EXISTS TURMA CASCADE;
DROP TABLE IF EXISTS DISCIPLINA CASCADE;
DROP TABLE IF EXISTS ALUNO CASCADE;
DROP TABLE IF EXISTS PROFESSOR CASCADE;
DROP TABLE IF EXISTS CARDAPIO_DIA CASCADE;
DROP TABLE IF EXISTS RESTAURANTE_UNIVERSITARIO CASCADE;
DROP TABLE IF EXISTS DEPARTAMENTO CASCADE;
DROP TABLE IF EXISTS PESSOA CASCADE;

CREATE TABLE PESSOA (
    cpf VARCHAR(14) PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(100),
    data_nascimento DATE, 
    senha VARCHAR(50) NOT NULL,
    saldo DECIMAL(10,2) DEFAULT 15.00
);

CREATE TABLE DEPARTAMENTO (
    cod_departamento INT PRIMARY KEY,
    nome_departamento VARCHAR(100) NOT NULL
);

CREATE TABLE RESTAURANTE_UNIVERSITARIO (
    codigo_ru INT PRIMARY KEY,
    nome VARCHAR(100),
    capacidade INT
);

CREATE TABLE CARDAPIO_DIA (
    id_cardapio INT PRIMARY KEY,
    prato_principal VARCHAR(100),
    sobremesa VARCHAR(100)
);

CREATE TABLE PROFESSOR (
    matricula_professor INT PRIMARY KEY,
    salario DECIMAL(10,2),
    fk_cod_departamento INT,
    fk_cpf VARCHAR(14) NOT NULL,
    FOREIGN KEY (fk_cod_departamento) REFERENCES DEPARTAMENTO(cod_departamento),
    FOREIGN KEY (fk_cpf) REFERENCES PESSOA(cpf)
);

CREATE TABLE ALUNO (
    matricula_aluno INT PRIMARY KEY,
    curso VARCHAR(100),
    fk_cpf VARCHAR(14) NOT NULL,
    FOREIGN KEY (fk_cpf) REFERENCES PESSOA(cpf)
);

CREATE TABLE DISCIPLINA (
    cod_disciplina INT PRIMARY KEY,
    nome_disciplina VARCHAR(100),
    creditos INT,
    fk_cod_departamento INT,
    fk_matricula_professor INT,
    FOREIGN KEY (fk_matricula_professor) REFERENCES PROFESSOR(matricula_professor),
    FOREIGN KEY (fk_cod_departamento) REFERENCES DEPARTAMENTO(cod_departamento)
);

CREATE TABLE TURMA (
    cod_turma VARCHAR(10),
    fk_cod_disciplina INT,
    semestre VARCHAR(10),
    horario VARCHAR(50),
    fk_matricula_prof INT,
    capacidade INT DEFAULT 30,
    PRIMARY KEY (cod_turma, fk_cod_disciplina),
    FOREIGN KEY (fk_cod_disciplina) REFERENCES DISCIPLINA(cod_disciplina),
    FOREIGN KEY (fk_matricula_prof) REFERENCES PROFESSOR(matricula_professor)
);

CREATE TABLE MATRICULA (
    fk_matricula_aluno INT,
    fk_cod_disciplina INT,
    fk_cod_turma VARCHAR(10),
    frequencia DECIMAL(5,2),
    PRIMARY KEY (fk_matricula_aluno, fk_cod_disciplina, fk_cod_turma),
    FOREIGN KEY (fk_matricula_aluno) REFERENCES ALUNO(matricula_aluno),
    FOREIGN KEY (fk_cod_turma, fk_cod_disciplina) REFERENCES TURMA(cod_turma, fk_cod_disciplina)
);

CREATE TABLE AVALIACAO (
    id_avaliacao VARCHAR(10), 
    fk_matricula_aluno INT,
    fk_cod_disciplina INT,
    fk_cod_turma VARCHAR(10),
    nota DECIMAL(4,2),
    PRIMARY KEY (id_avaliacao, fk_matricula_aluno, fk_cod_disciplina, fk_cod_turma),
    FOREIGN KEY (fk_matricula_aluno, fk_cod_disciplina, fk_cod_turma) 
        REFERENCES MATRICULA(fk_matricula_aluno, fk_cod_disciplina, fk_cod_turma)
);

CREATE TABLE REGISTRO_USO (
    fk_cpf VARCHAR(14),
    fk_codigo_ru INT,
    data_hora TIMESTAMP, 
    fk_id_cardapio INT NOT NULL, 
    PRIMARY KEY (fk_cpf, fk_codigo_ru, data_hora),
    FOREIGN KEY (fk_cpf) REFERENCES PESSOA(cpf),
    FOREIGN KEY (fk_codigo_ru) REFERENCES RESTAURANTE_UNIVERSITARIO(codigo_ru),
    FOREIGN KEY (fk_id_cardapio) REFERENCES CARDAPIO_DIA(id_cardapio)
);

CREATE TABLE PESSOAS_EXCLUIDAS (
    id SERIAL PRIMARY KEY,
    matricula VARCHAR(20),
    cpf VARCHAR(14),
    tipo VARCHAR(50),
    nome VARCHAR(100),
    data_exclusao TIMESTAMP DEFAULT NOW()
);

CREATE OR REPLACE FUNCTION salvar_aluno_excluido()
RETURNS TRIGGER AS $$
DECLARE
    v_nome_pessoa VARCHAR(100);
BEGIN
    SELECT nome INTO v_nome_pessoa 
    FROM PESSOA 
    WHERE cpf = OLD.fk_cpf; 

    INSERT INTO PESSOAS_EXCLUIDAS (matricula, cpf, tipo, nome)
    VALUES (
        OLD.matricula_aluno::VARCHAR, 
        OLD.fk_cpf,                   
        'Aluno',
        v_nome_pessoa                 
    );
    RETURN OLD;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION salvar_professor_excluido()
RETURNS TRIGGER AS $$
DECLARE
    v_nome_pessoa VARCHAR(100);
BEGIN
    SELECT nome INTO v_nome_pessoa 
    FROM PESSOA 
    WHERE cpf = OLD.fk_cpf; 

    INSERT INTO PESSOAS_EXCLUIDAS (matricula, cpf, tipo, nome)
    VALUES (
        OLD.matricula_professor::VARCHAR, 
        OLD.fk_cpf,                       
        'Professor',
        v_nome_pessoa
    );
    RETURN OLD;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_log_aluno_exclusao
AFTER DELETE ON ALUNO
FOR EACH ROW
EXECUTE FUNCTION salvar_aluno_excluido();

CREATE TRIGGER trigger_log_professor_exclusao
AFTER DELETE ON PROFESSOR
FOR EACH ROW
EXECUTE FUNCTION salvar_professor_excluido();

-- 4. CONSULTAS AVANÇADAS (REQUISITOS FUNCIONAIS)
-- ==========================================================

-- 4.1. Consulta com Quantificador ALL (Alunos Excepcionais)
-- Lista a maior nota de um aluno APENAS SE ela for maior ou igual a TODAS as outras notas da mesma turma.
-- Utiliza uma CTE (Common Table Expression) para filtrar e depois agrupa para limpar duplicatas.

WITH QuantifiedResults AS (
    SELECT
        G1.fk_matricula_aluno AS matricula_aluno,
        G1.fk_cod_disciplina AS cod_disciplina,
        G1.nota AS nota_obtida
    FROM AVALIACAO G1
    JOIN TURMA T ON 
        T.fk_cod_disciplina = G1.fk_cod_disciplina AND 
        T.cod_turma = G1.fk_cod_turma
    
    WHERE T.fk_matricula_prof = %s 
    
    AND G1.nota >= ALL ( 
        SELECT G2.nota
        FROM AVALIACAO G2
        WHERE G2.fk_cod_disciplina = G1.fk_cod_disciplina 
          AND G2.fk_matricula_aluno != G1.fk_matricula_aluno
    )
)
-- Agrupamento para limpar duplicatas (pega a maior nota se o aluno tiver N1 e N2 altas)
SELECT
    P.nome,
    QR.matricula_aluno,
    MAX(QR.nota_obtida) AS melhor_nota_qualificada,
    D.nome_disciplina
FROM QuantifiedResults QR
JOIN ALUNO A ON A.matricula_aluno = QR.matricula_aluno
JOIN PESSOA P ON P.cpf = A.fk_cpf
JOIN DISCIPLINA D ON D.cod_disciplina = QR.cod_disciplina
GROUP BY P.nome, QR.matricula_aluno, D.nome_disciplina
ORDER BY melhor_nota_qualificada DESC;


-- 4.2. Consulta com GROUP BY e HAVING
-- Lista a média geral das disciplinas, mas APENAS aquelas que possuem notas lançadas.
SELECT 
    D.nome_disciplina,
    AVG(AV.nota) as media_geral
FROM DISCIPLINA D
JOIN TURMA T ON T.fk_cod_disciplina = D.cod_disciplina
LEFT JOIN AVALIACAO AV ON AV.fk_cod_disciplina = D.cod_disciplina
GROUP BY D.nome_disciplina
HAVING MAX(AV.nota) IS NOT NULL;


-- 4.3. Consulta com Múltiplos JOINs (Boletim Geral)
-- Relatório completo cruzando 5 tabelas para gerar a média final e frequência.
SELECT 
    P.nome,
    D.nome_disciplina,
    M.frequencia,
    AVG(AV.nota) as media_final
FROM MATRICULA M
JOIN ALUNO A ON M.fk_matricula_aluno = A.matricula_aluno
JOIN PESSOA P ON A.fk_cpf = P.cpf
JOIN DISCIPLINA D ON M.fk_cod_disciplina = D.cod_disciplina
JOIN AVALIACAO AV ON M.fk_matricula_aluno = AV.fk_matricula_aluno
GROUP BY P.nome, D.nome_disciplina, M.frequencia
ORDER BY P.nome ASC;
